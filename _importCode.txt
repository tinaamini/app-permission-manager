چک کردن امضا APK:
keytool -printcert -jarfile build\app\outputs\flutter-apk\app-release.apk


دستور نصب Release APK:
flutter install --use-application-binary build\app\outputs\flutter-apk\app-release.apk
adb install -r build\app\outputs\flutter-apk\app-release.apk

creat key.jks 
keytool -genkey -v -keystore key.jks -keyalg RSA -keysize 2048 -validity 10000 -alias my-key-alias
bundle :
java -jar bundlesigner-0.1.13.jar genbin --bundle app-release.aab --bin ./ --v2-signing-enabled true --v3-signing-enabled true --ks key.jks -v

dart run flutter_launcher_icons

set splash :
dart run flutter_native_splash:create
